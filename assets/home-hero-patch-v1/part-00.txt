async function fetchBytes(url, retries = 5) {
  let lastError;
  for (let attempt = 1; attempt <= retries; attempt += 1) {
    try {
      const response = await fetch(url, { headers: requestHeaders, redirect: 'follow' });
      if (!response.ok) throw new Error(`HTTP ${response.status} ${url}`);
      return {
        data: Buffer.from(await response.arrayBuffer()),
        type: (response.headers.get('content-type') || '').split(';')[0].toLowerCase(),
        finalUrl: response.url,
      };
    } catch (error) {
      lastError = error;
      if (attempt < retries) await new Promise(resolve => setTimeout(resolve, 900 * attempt));
    }
  }
  throw lastError;
}

function localFile(urlPath) {
  const clean = decodeURIComponent(urlPath).replace(/^\/+/, '');
  if (!clean || clean.includes('..')) throw new Error(`Unsafe path: ${urlPath}`);
  return path.join(DIST, clean);
}

function writeAsset(urlPath, data) {
  const file = localFile(urlPath);
  fs.mkdirSync(path.dirname(file), { recursive: true });
  fs.writeFileSync(file, data);
}

function normalizeAsset(raw, baseUrl = SOURCE) {
  if (!raw) return null;
  const value = raw.trim();
  if (!value || /^(?:#|data:|blob:|javascript:|mailto:|tel:)/i.test(value)) return null;
  let url;
  try { url = new URL(value, baseUrl); } catch { return null; }
  if (url.host !== SOURCE_HOST || url.pathname === '/' || url.pathname.endsWith('/')) return null;
  url.hash = '';
  return url;
}

function escapeHtml(value) {
  return String(value)
    .replaceAll('&', '&amp;')
    .replaceAll('"', '&quot;')
    .replaceAll("'", '&#39;')
    .replaceAll('<', '&lt;')
    .replaceAll('>', '&gt;');
}

function stripTags(value) {
  return value
    .replace(/<script\b[^>]*>[\s\S]*?<\/script>/gi, '')
    .replace(/<style\b[^>]*>[\s\S]*?<\/style>/gi, '')
    .replace(/<[^>]+>/g, '')
    .replace(/&nbsp;/gi, ' ')
    .replace(/&amp;/gi, '&')
    .replace(/\s+/g, ' ')
    .trim();
}

function findHrefByText(html, text) {
  const anchorPattern = /<a\b([^>]*)>([\s\S]*?)<\/a>/gi;
  for (const match of html.matchAll(anchorPattern)) {
    const label = stripTags(match[2]);
    if (!label.includes(text)) continue;
    const hrefMatch = match[1].match(/\bhref\s*=\s*(["'])(.*?)\1/i);
    if (hrefMatch?.[2]) return hrefMatch[2];
  }
  return null;
}

function externalAttrs(href) {
  return /^https?:\/\//i.test(href) ? ' target="_blank" rel="noopener"' : '';
}
